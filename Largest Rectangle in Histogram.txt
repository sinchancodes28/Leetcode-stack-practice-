class Solution {
    public int largestRectangleArea(int[] heights) {
        // for largest rectangle we need to calculate the area for each bar
        // area of rectangle = height * width 
        // height is constant for each bar 
        // we need to define the width 
        // width = right[i] - left[i] -1
        // where left and right are the nearset smallest boundarys
        Stack<Integer> st = new Stack<>();
        int width = 0;
        int ans = 0;
        int n = heights.length;
        int[] left = new int[n];
        int[] right = new int[n];
        // left nearest smallest
        for( int i = 0 ; i < n ; i++){
             while(!st.isEmpty() && heights[st.peek()] >= heights[i]){
                st.pop();
             }
        left[i]  =  st.isEmpty() ?  -1 : st.peek();
           st.push(i);
        }
        while(!st.isEmpty()){
            st.pop();
        }
          for( int  i = n-1 ; i >= 0 ; i--){
             while(!st.isEmpty() && heights[st.peek()] >= heights[i]){
                st.pop();
             }
        right[i]  =  st.isEmpty() ? n : st.peek();
           st.push(i);
        }
        for( int i = 0 ; i < n ; i++){
             width = right[i] - left[i] -1;
        int curr = heights[i] * width;
        ans = Math.max(ans , curr);
        }
        return ans;
    }
}
